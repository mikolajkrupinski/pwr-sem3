DELIMITER $$
CREATE FUNCTION kontrol(pesel char(11)) RETURNS int DETERMINISTIC
BEGIN

DECLARE k, i, v, w int DEFAULT 0;
WHILE (i<10) 
DO
    SET v=CAST(RIGHT(LEFT(pesel, (i+1)),1) AS int);
    IF ((i%4) = 0)
    THEN
        SET w=1;
    ELSEIF ((i%4) = 1) 
    THEN
        SET w=3;
    ELSEIF ((i%4) = 2) 
    THEN
        SET w=7;
    ELSE
        SET w=9;
    END IF;
    SET k = k + v*w;
    SET i = i +1;
END WHILE;
SET k = k % 10;
SET i = (i - k);
SET i = i % 10;
RETURN i;

END$$
DELIMITER ;

DELIMITER $$
CREATE FUNCTION czyPESELok(pesel char(11), data date, p char) RETURNS int DETERMINISTIC
BEGIN

DECLARE wynik int DEFAULT 0;
DECLARE rok, miesiac, dzien int;
DECLARE plec_cyfra int;
DECLARE po2000 int DEFAULT 0;

IF RIGHT(pesel, 1) <> (SELECT kontrol(pesel)) THEN
	RETURN wynik;
END IF;

SET rok = CAST(LEFT(pesel, 2) AS int);
SET miesiac = CAST(RIGHT(LEFT(pesel, 4), 2) AS int);
SET dzien = CAST(RIGHT(LEFT(pesel, 6), 2) AS int);

IF miesiac > 20 THEN
	SET miesiac = miesiac - 20;
	SET po2000 = 1;
	IF rok + 2000 <> YEAR(data) OR miesiac <> MONTH(data) OR dzien <> DAY(data) THEN
		RETURN wynik;
	END IF;
END IF;

IF po2000 = 0 AND (rok + 1900 <> YEAR(data) OR miesiac <> MONTH(data) OR dzien <> DAY(data)) THEN
	RETURN wynik;
END IF;

SET plec_cyfra = CAST(LEFT(RIGHT(pesel, 2), 1) AS int);
IF (plec_cyfra % 2 = 0 AND p = 'M') OR (plec_cyfra % 2 = 1 AND p = 'K') THEN
	RETURN wynik;
END IF;

SET wynik = 1;
RETURN wynik;

END$$
DELIMITER ;

DELIMITER $$
CREATE PROCEDURE stworzPracownikow() BEGIN
DECLARE done INT DEFAULT FALSE;
DECLARE id, z, pmin, pmax, pens, wiek INT;
DECLARE d DATE;
DECLARE p CHAR;
DECLARE czlowiek CURSOR FOR
(SELECT ludzie_id, data_urodzenia, plec FROM Ludzie);
DECLARE CONTINUE HANDLER FOR NOT FOUND SET done = TRUE;
OPEN czlowiek;

dodaj: LOOP
	FETCH czlowiek INTO id, d, p;
	IF done THEN
		LEAVE dodaj;
	END IF;
	SET wiek = TIMESTAMPDIFF(YEAR, d, CURDATE());
	IF (P='K' AND wiek > 60) OR (P='M' AND wiek > 65) THEN
		SELECT zawod_id, pensja_min, pensja_max INTO z, pmin, pmax FROM Zawody WHERE zawod_id != 3 ORDER BY RAND() LIMIT 1;
		SET pens = FLOOR(pmin + RAND() * (pmax-pmin+1));
		INSERT INTO Pracownicy VALUES (id, z, pens);
	ELSE
		IF wiek > 18 THEN
			SELECT zawod_id, pensja_min, pensja_max INTO z, pmin, pmax FROM Zawody ORDER BY RAND() LIMIT 1;
			SET pens = FLOOR(pmin + RAND() * (pmax-pmin+1));
			INSERT INTO Pracownicy VALUES (id, z, pens);
		END IF;
	END IF;
END LOOP;
END $$
DELIMITER ;

CALL stworzPracownikow();

DECLARE count int;
DECLARE z_id int;

IF NOT EXISTS(SELECT * FROM Zawody Z JOIN Pracownicy P ON Z.zawod_id = P.zawod_id WHERE Z.nazwa = n AND P.pensja * 1.05 > Z.pensja_max) THEN

END$$
