2.
CREATE TABLE Producent (
	ID int unsigned NOT NULL AUTO_INCREMENT,
	nazwa varchar(50),
    	kraj varchar(20),
    	PRIMARY KEY (ID) );

CREATE TABLE Obiektyw (
    	ID int unsigned NOT NULL AUTO_INCREMENT,
    	model varchar(30) NOT NULL,
    	minPrzeslona float unsigned NOT NULL,
    	maxPrzeslona float unsigned NOT NULL,
    	PRIMARY KEY (ID),
    	CHECK (minPrzeslona < maxPrzeslona) );

CREATE TABLE Matryca (
    	ID int unsigned NOT NULL AUTO_INCREMENT,
   	przekatna decimal(4,2) unsigned NOT NULL,
    	rozdzielczosc decimal(3,1) unsigned NOT NULL,
    	typ varchar(10) NOT NULL,
    	PRIMARY KEY (ID) );
ALTER TABLE Matryca AUTO_INCREMENT=100;

CREATE TABLE Aparat (
    	model varchar(30) NOT NULL,
    	producent int unsigned NOT NULL,
   	matryca int unsigned NOT NULL,
    	obiektyw int unsigned NOT NULL,
    	typ enum ('kompaktowy', 'lustrzanka', 'profesjonalny', 'inny') NOT NULL,
    	PRIMARY KEY (model),
    	FOREIGN KEY (producent) REFERENCES Producent(ID),
    	FOREIGN KEY (matryca) REFERENCES Matryca(ID),
    	FOREIGN KEY (obiektyw) REFERENCES Obiektyw(ID) );

DELIMITER $$
CREATE PROCEDURE IF NOT EXISTS wypelnijAparat ()
BEGIN

DECLARE i int;
DECLARE mat int;
DECLARE ob int;
DECLARE prod int;
DECLARE t int;
DECLARE model_id int;

SET i = 1;
WHILE i <= 100 DO
	SELECT ID INTO mat FROM Matryca ORDER BY RAND() LIMIT 1;
	SELECT ID INTO ob FROM Obiektyw ORDER BY RAND() LIMIT 1;
	SELECT ID INTO prod FROM Producent ORDER BY RAND() LIMIT 1;

	SET t = ROUND(RAND() * 3 + 1);

	SELECT COUNT(*) INTO model_id FROM Aparat;

	INSERT INTO Aparat (model, producent, matryca, obiektyw, typ)
		VALUES (CONCAT('aparat', model_id + 1), prod, mat, ob, t);
	
	SET i = i + 1;
END WHILE;

END$$
DELIMITER ;

CALL wypelnijAparat();

DELIMITER $$
CREATE FUNCTION IF NOT EXISTS znajdzProducenta(idProd int) RETURNS varchar(30)
BEGIN

DECLARE wynik varchar(30);
SELECT a.model INTO wynik FROM Aparat a JOIN Producent p ON p.ID = a.producent JOIN Matryca m ON m.ID = a.matryca WHERE p.ID = idProd ORDER BY m.przekatna LIMIT 1;
RETURN wynik;

END$$
DELIMITER ;

CREATE TRIGGER IF NOT EXISTS dodajProducenta BEFORE INSERT ON Aparat FOR EACH ROW
BEGIN

	INSERT INTO Producent (ID) VALUE (NEW.producent);
END IF;

END$$

DELIMITER $$
CREATE TRIGGER IF NOT EXISTS liczbaAparatowDelete AFTER DELETE ON Aparat FOR EACH ROW
BEGIN

UPDATE Producent SET liczbaModeli = (SELECT COUNT(*) FROM Aparat WHERE Aparat.producent = Producent.ID) WHERE ID = OLD.producent;

END$$
DELIMITER ;



DELIMITER $$
CREATE TRIGGER IF NOT EXISTS liczbaAparatowInsert AFTER INSERT ON Aparat FOR EACH ROW
BEGIN

UPDATE Producent SET liczbaModeli = (SELECT COUNT(*) FROM Aparat WHERE Aparat.producent = Producent.ID) WHERE ID = NEW.producent;

END$$
DELIMITER ;



DELIMITER $$
CREATE TRIGGER IF NOT EXISTS liczbaAparatowUpdate AFTER UPDATE ON Aparat FOR EACH ROW
BEGIN

UPDATE Producent SET liczbaModeli = (SELECT COUNT(*) FROM Aparat WHERE Aparat.producent = Producent.ID) WHERE ID = OLD.producent OR ID = NEW.producent;

END$$
DELIMITER ;
